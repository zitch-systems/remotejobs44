// RemoteJobs44 — extended screens (part 2): plans/paywall, settings, AI toolkit.
import React, { useState } from 'react';
import { View, Text, ScrollView, Pressable, Switch, Alert, Appearance, Linking, TextInput } from 'react-native';
import { SafeAreaView } from 'react-native-safe-area-context';
import { useStore } from './store';
import { AI_TOOLS, FREE_FEATURES, PRO_FEATURES, PRO_PRICE } from './data';
import { Icon } from './ui';
import { palette } from './theme';
import { sx } from './screens';
import { TopBar } from './extra';

export function PlansScreen({ navigation }: any) {
  const { theme, state, upgrade, cancelPro } = useStore(); const s = sx(theme);
  const [cycle, setCycle] = useState<'monthly' | 'yearly'>('monthly');
  const price = cycle === 'monthly' ? PRO_PRICE.monthly : PRO_PRICE.yearly;
  const per = cycle === 'monthly' ? '/month' : '/year';
  const doUpgrade = () => { upgrade(); Alert.alert('Welcome to Pro 🎉', 'Unlimited applications and all AI tools are unlocked.'); navigation.goBack(); };
  return (
    <View style={{ flex: 1, backgroundColor: theme.bg }}>
      <SafeAreaView edges={['top']} style={{ backgroundColor: palette.ocean2 }}>
        <View style={{ paddingHorizontal: 8, paddingTop: 4 }}>
          <Pressable onPress={() => navigation.goBack()} hitSlop={8} style={{ width: 40, height: 40, alignItems: 'center', justifyContent: 'center' }}><Icon name="x" size={22} color="#fff" /></Pressable>
        </View>
        <View style={{ alignItems: 'center', paddingHorizontal: 24, paddingBottom: 26 }}>
          <View style={{ width: 64, height: 64, borderRadius: 20, backgroundColor: 'rgba(255,255,255,0.1)', borderWidth: 1, borderColor: 'rgba(255,255,255,0.2)', alignItems: 'center', justifyContent: 'center', marginBottom: 14 }}><Icon name="crown" size={30} color="#fde68a" /></View>
          <Text style={{ color: '#fff', fontWeight: '800', fontSize: 24, textAlign: 'center' }}>Go unlimited with Pro</Text>
          <Text style={{ color: '#aebfd6', fontSize: 14, textAlign: 'center', marginTop: 8, lineHeight: 20 }}>Apply to as many roles as you want — and let AI do the heavy lifting.</Text>
        </View>
      </SafeAreaView>
      <ScrollView contentContainerStyle={{ padding: 18, paddingBottom: 40 }}>
        {state.pro ? (
          <View style={[s.cardBlock, { padding: 18, alignItems: 'center' }]}>
            <View style={{ width: 48, height: 48, borderRadius: 24, backgroundColor: theme.success, alignItems: 'center', justifyContent: 'center' }}><Icon name="check" size={26} color="#fff" /></View>
            <Text style={{ fontWeight: '800', fontSize: 16, color: theme.fg1, marginTop: 10 }}>You're on Pro</Text>
            <Text style={{ color: theme.fg3, fontSize: 13, marginTop: 4, textAlign: 'center' }}>Thanks for supporting RemoteJobs44.</Text>
            <Pressable onPress={() => { cancelPro(); Alert.alert('Pro cancelled', 'You\u2019re back on the Free plan.'); }} style={{ marginTop: 16 }}><Text style={{ color: theme.fg4, fontWeight: '700' }}>Cancel subscription</Text></Pressable>
          </View>
        ) : (
          <>
            <View style={{ flexDirection: 'row', backgroundColor: theme.card, borderWidth: 1.5, borderColor: theme.line2, borderRadius: 13, padding: 4, marginBottom: 18 }}>
              {(['monthly', 'yearly'] as const).map(c => (
                <Pressable key={c} onPress={() => setCycle(c)} style={{ flex: 1, paddingVertical: 9, borderRadius: 10, alignItems: 'center', backgroundColor: cycle === c ? theme.brand600 : 'transparent' }}>
                  <Text style={{ fontWeight: '700', fontSize: 13, color: cycle === c ? '#fff' : theme.fg3 }}>{c === 'monthly' ? 'Monthly' : 'Yearly · save 34%'}</Text>
                </Pressable>
              ))}
            </View>
            <View style={[s.cardBlock, { borderColor: theme.brand600, borderWidth: 1.5, padding: 18 }]}>
              <View style={{ flexDirection: 'row', alignItems: 'flex-end', gap: 5 }}>
                <Text style={{ fontWeight: '800', fontSize: 34, color: theme.fg1, letterSpacing: -1 }}>{price}</Text>
                <Text style={{ color: theme.fg3, fontSize: 14, marginBottom: 6 }}>{per}</Text>
              </View>
              <View style={{ marginTop: 14, gap: 11 }}>
                {PRO_FEATURES.map(f => (
                  <View key={f} style={{ flexDirection: 'row', gap: 10, alignItems: 'flex-start' }}><Icon name="check" size={18} color={theme.success} /><Text style={{ flex: 1, fontSize: 14, color: theme.fg2, lineHeight: 20 }}>{f}</Text></View>
                ))}
              </View>
              <Pressable onPress={doUpgrade} style={[s.btn, s.btnPrimary, { marginTop: 18 }]}><Icon name="crown" size={18} color="#fff" /><Text style={s.btnPrimaryTx}>Upgrade now</Text></Pressable>
              <Text style={{ textAlign: 'center', color: theme.fg4, fontSize: 11, marginTop: 10 }}>Cancel anytime · or grab a {PRO_PRICE.daily} day pass</Text>
            </View>
            <Text style={{ color: theme.fg4, fontWeight: '700', fontSize: 12, marginTop: 22, marginBottom: 10, textTransform: 'uppercase', letterSpacing: 0.5 }}>Free plan includes</Text>
            <View style={{ gap: 9 }}>
              {FREE_FEATURES.map(f => <View key={f} style={{ flexDirection: 'row', gap: 10 }}><Icon name="check" size={16} color={theme.fg4} /><Text style={{ color: theme.fg3, fontSize: 13.5 }}>{f}</Text></View>)}
            </View>
          </>
        )}
      </ScrollView>
    </View>
  );
}

export function SettingsScreen({ navigation }: any) {
  const { theme, state, setTheme, signOut, resetData } = useStore(); const s = sx(theme);
  const [push, setPush] = useState(true); const [emails, setEmails] = useState(true);
  const matchSystem = () => setTheme(Appearance.getColorScheme() === 'dark' ? 'dark' : 'light');
  const lbl = { fontSize: 12, fontWeight: '700' as const, color: theme.fg4, textTransform: 'uppercase' as const, letterSpacing: 0.5, marginTop: 18, marginBottom: 8 };
  const confirmDelete = () => Alert.alert('Delete account?', 'This permanently removes your data on this device.', [{ text: 'Cancel', style: 'cancel' }, { text: 'Delete', style: 'destructive', onPress: () => { resetData(); signOut(); } }]);
  const Group = ({ items }: any) => (
    <View style={[s.cardBlock, { overflow: 'hidden' }]}>
      {items.map((it: any, i: number) => (
        <Pressable key={it.label} onPress={it.onPress} style={{ flexDirection: 'row', alignItems: 'center', gap: 12, padding: 14, borderTopWidth: i ? 1 : 0, borderTopColor: theme.line }}>
          <Icon name={it.icon} size={19} color={it.danger ? '#e11d48' : theme.fg3} />
          <Text style={{ flex: 1, fontSize: 14.5, color: it.danger ? '#e11d48' : theme.fg1 }}>{it.label}</Text>
          {it.chevron === false ? null : <Icon name="chevron" size={18} color={theme.fg4} />}
        </Pressable>
      ))}
    </View>
  );
  return (
    <SafeAreaView edges={['top']} style={{ flex: 1, backgroundColor: theme.bg }}>
      <TopBar title="Settings" nav={navigation} />
      <ScrollView contentContainerStyle={{ padding: 18, paddingBottom: 40 }}>
        <Text style={lbl}>Appearance</Text>
        <View style={{ flexDirection: 'row', backgroundColor: theme.card, borderWidth: 1.5, borderColor: theme.line2, borderRadius: 13, padding: 4 }}>
          {(['light', 'dark'] as const).map(k => (
            <Pressable key={k} onPress={() => setTheme(k)} style={{ flex: 1, paddingVertical: 9, borderRadius: 10, alignItems: 'center', backgroundColor: state.theme === k ? theme.brand600 : 'transparent' }}>
              <Text style={{ fontWeight: '700', fontSize: 13.5, color: state.theme === k ? '#fff' : theme.fg3 }}>{k === 'light' ? 'Light' : 'Dark'}</Text>
            </Pressable>
          ))}
          <Pressable onPress={matchSystem} style={{ flex: 1, paddingVertical: 9, borderRadius: 10, alignItems: 'center' }}><Text style={{ fontWeight: '700', fontSize: 13.5, color: theme.fg3 }}>System</Text></Pressable>
        </View>
        <Text style={lbl}>Notifications</Text>
        <View style={[s.cardBlock, { overflow: 'hidden' }]}>
          <View style={{ flexDirection: 'row', alignItems: 'center', padding: 14 }}><Text style={{ flex: 1, fontSize: 14.5, color: theme.fg1 }}>Push notifications</Text><Switch value={push} onValueChange={setPush} trackColor={{ true: theme.brand600 }} /></View>
          <View style={{ flexDirection: 'row', alignItems: 'center', padding: 14, borderTopWidth: 1, borderTopColor: theme.line }}><Text style={{ flex: 1, fontSize: 14.5, color: theme.fg1 }}>Email job alerts</Text><Switch value={emails} onValueChange={setEmails} trackColor={{ true: theme.brand600 }} /></View>
        </View>
        <Text style={lbl}>Account</Text>
        <Group items={[
          { icon: 'user', label: 'Edit profile', onPress: () => navigation.navigate('EditProfile') },
          { icon: 'filter', label: 'Job preferences', onPress: () => navigation.navigate('Preferences') },
          { icon: 'lock', label: 'Change password', onPress: () => navigation.navigate('Forgot') },
        ]} />
        <Text style={lbl}>Support</Text>
        <Group items={[
          { icon: 'help', label: 'Help & support', onPress: () => Linking.openURL('mailto:support@remotejobs44.com') },
          { icon: 'shield', label: 'Privacy policy', onPress: () => Linking.openURL('https://remotejobs44.com/privacy') },
          { icon: 'doc', label: 'Terms of service', onPress: () => Linking.openURL('https://remotejobs44.com/terms') },
        ]} />
        <View style={{ marginTop: 18 }}>
          <Group items={[
            { icon: 'logout', label: 'Sign out', onPress: signOut, chevron: false },
            { icon: 'trash', label: 'Delete account', onPress: confirmDelete, danger: true, chevron: false },
          ]} />
        </View>
        <Text style={{ textAlign: 'center', color: theme.fg4, fontSize: 11.5, marginTop: 20 }}>RemoteJobs44 · v1.0.0 · Deep Ocean</Text>
      </ScrollView>
    </SafeAreaView>
  );
}

export function AIToolsScreen({ navigation }: any) {
  const { theme, state } = useStore(); const s = sx(theme);

    <SafeAreaView edges={['top']} style={{ flex: 1, backgroundColor: theme.bg }}>

            <View style={{ width: 46, height: 46, borderRadius: 13, backgroundColor: theme.brand50, alignItems: 'center', justifyContent: 'center' }}><Icon name={t.icon} size={22} color={theme.brand600} /></View>
            <View style={{ flex: 1 }}>
              <Text style={{ fontWeight: '700', fontSize: 15, color: theme.fg1 }}>{t.title}</Text>
              <Text style={{ fontSize: 12.5, color: theme.fg3, marginTop: 2 }}>{t.blurb}</Text>
            </View>
            <Icon name={state.pro ? 'chevron' : 'lock'} size={state.pro ? 18 : 16} color={theme.fg4} />
          </Pressable>
        ))}
      </ScrollView>
    </SafeAreaView>
  );
}

export function AIToolScreen({ route, navigation }: any) {
  const { theme } = useStore(); const s = sx(theme);
  const tool = AI_TOOLS.find(t => t.key === route.params?.key) || AI_TOOLS[0];
  const [busy, setBusy] = useState(false); const [done, setDone] = useState(false);
  const generate = () => { setBusy(true); setDone(false); setTimeout(() => { setBusy(false); setDone(true); }, 1100); };
  return (
    <SafeAreaView edges={['top']} style={{ flex: 1, backgroundColor: theme.bg }}>
      <TopBar title={tool.title} nav={navigation} />
      <ScrollView contentContainerStyle={{ padding: 18, paddingBottom: 40 }}>
        <View style={{ flexDirection: 'row', alignItems: 'center', gap: 13, marginBottom: 18 }}>
          <View style={{ width: 52, height: 52, borderRadius: 15, backgroundColor: theme.brand50, alignItems: 'center', justifyContent: 'center' }}><Icon name={tool.icon} size={24} color={theme.brand600} /></View>
          <Text style={{ flex: 1, fontSize: 14, color: theme.fg2, lineHeight: 20 }}>{tool.blurb}</Text>
        </View>
        {!done ? (
          <Pressable onPress={generate} disabled={busy} style={[s.btn, s.btnPrimary, busy ? { opacity: 0.6 } : null]}><Icon name="spark" size={18} color="#fff" /><Text style={s.btnPrimaryTx}>{busy ? 'Working…' : tool.cta}</Text></Pressable>
        ) : (
          <View style={[s.cardBlock, { padding: 16 }]}>
            <View style={{ flexDirection: 'row', alignItems: 'center', gap: 8, marginBottom: 12 }}><Icon name="spark" size={16} color={theme.brand600} /><Text style={{ fontWeight: '800', fontSize: 13, color: theme.brand600 }}>AI result</Text></View>
            {tool.result.map((line, i) => (
              <View key={i} style={{ flexDirection: 'row', gap: 10, marginBottom: 12 }}>
                {tool.key === 'cover' ? null : <View style={{ width: 6, height: 6, borderRadius: 3, backgroundColor: theme.brand500, marginTop: 7 }} />}
                <Text style={{ flex: 1, fontSize: 13.5, color: theme.fg2, lineHeight: 20 }}>{line}</Text>
              </View>
            ))}
            <Pressable onPress={generate} style={{ marginTop: 4 }}><Text style={{ color: theme.brand600, fontWeight: '700', fontSize: 13 }}>Regenerate</Text></Pressable>
          </View>
        )}
      </ScrollView>
    </SafeAreaView>
  );
}

export function ForgotScreen({ navigation }: any) {
  const { theme } = useStore(); const s = sx(theme);
  const [email, setEmail] = useState(''); const [sent, setSent] = useState(false);
  return (
    <SafeAreaView edges={['top']} style={{ flex: 1, backgroundColor: theme.bg }}>
      <TopBar title="Reset password" nav={navigation} />
      <ScrollView contentContainerStyle={{ padding: 18, paddingBottom: 28 }}>
        {sent ? (
          <View style={{ alignItems: 'center', paddingVertical: 30 }}>
            <View style={{ width: 64, height: 64, borderRadius: 20, backgroundColor: theme.brand50, alignItems: 'center', justifyContent: 'center', marginBottom: 16 }}><Icon name="mail" size={30} color={theme.brand600} /></View>
            <Text style={{ fontWeight: '800', fontSize: 18, color: theme.fg1 }}>Check your inbox</Text>
            <Text style={{ color: theme.fg3, fontSize: 14, marginTop: 8, textAlign: 'center', maxWidth: 280, lineHeight: 21 }}>We sent a reset link to {email || 'your email'}. (Simulated in this prototype.)</Text>
            <Pressable onPress={() => navigation.goBack()} style={[s.btn, s.btnPrimary, { marginTop: 20, alignSelf: 'stretch' }]}><Text style={s.btnPrimaryTx}>Back to settings</Text></Pressable>
          </View>
        ) : (
          <>
            <Text style={{ color: theme.fg2, fontSize: 14, lineHeight: 21, marginBottom: 16 }}>Enter your account email and we'll send you a link to reset your password.</Text>
            <View style={s.input}><Icon name="mail" size={18} color={theme.fg4} /><TextInputProxy value={email} onChangeText={setEmail} theme={theme} /></View>
            <Pressable onPress={() => setSent(true)} style={[s.btn, s.btnPrimary, { marginTop: 16 }]}><Text style={s.btnPrimaryTx}>Send reset link</Text></Pressable>
          </>
        )}
      </ScrollView>
    </SafeAreaView>
  );
}

function TextInputProxy({ value, onChangeText, theme }: any) {
  return <TextInput value={value} onChangeText={onChangeText} placeholder="you@email.com" placeholderTextColor={theme.fg4} autoCapitalize="none" keyboardType="email-address" style={{ flex: 1, fontSize: 15, color: theme.fg1 }} />;
}
